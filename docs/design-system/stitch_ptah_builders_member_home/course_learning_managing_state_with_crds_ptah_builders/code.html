Managing State with CRDs

schedule 22 mins | visibility 1.2k views

State Management Strategies

In a Kubernetes operator, the custom resource (CR) is the single source of truth. However, ensuring idempotency in your reconcile loop requires careful handling of the observed state versus the desired state defined in the CR spec.

  • Observed State: Read dynamically from the cluster (e.g., existing Pods, Services).
  • Desired State: Derived entirely from the Spec field of your CRD.
  • Status Subresource: Used strictly to report the diff between observed and desired state.
GO 
func (r *MyAppReconciler) Reconcile(ctx context.Context, req ctrl.Request) (ctrl.Result, error) {
    log := log.FromContext(ctx)

    // Fetch the MyApp instance
    var myApp v1alpha1.MyApp
    if err := r.Get(ctx, req.NamespacedName, &myApp); err != nil {
        return ctrl.Result{}, client.IgnoreNotFound(err)
    }

    // Example: Update Status
    myApp.Status.Phase = "Reconciling"
    if err := r.Status().Update(ctx, &myApp); err != nil {
        log.Error(err, "Failed to update MyApp status")
        return ctrl.Result{}, err
    }

    return ctrl.Result{}, nil
}
