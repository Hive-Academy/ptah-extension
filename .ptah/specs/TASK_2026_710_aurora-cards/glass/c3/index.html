I'll extract the check into a SessionGuard, register it on the auth routes and leave the middleware only to attach the request id. This is the guard: 
@Injectable()
export class SessionGuard implements CanActivate {
  constructor(private readonly sessions: SessionService) {}
  async canActivate(ctx: ExecutionContext) {
    const req = ctx.switchToHttp().getRequest();
    // 401 contract unchanged
    if (!(await this.sessions.verify(req.headers.authorization)))
      throw new UnauthorizedException('Invalid session');
    return true;  }  }
