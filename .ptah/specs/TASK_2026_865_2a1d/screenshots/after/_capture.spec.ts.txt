const OLD_SESSION = '3f2504e0-4f89-41d3-9a0c-0305e82c3301';

function turnStateIdle(tabId: string, sessionId: string) {
  return {
    type: 'chat:chunk',
    payload: {
      tabId, sessionId,
      event: {
        id: randomUUID(), eventType: 'turn_state', timestamp: Date.now(), sessionId,
        messageId: `turn-state-${tabId}-1`, phase: 'idle', revision: 1,
        backgroundTasks: [], sessionCrons: [], terminalReason: 'completed',
      },
    },
  };
}

async function setTheme(ui: any, rpcBridge: any, theme: string) {
  await rpcBridge.setState({ 'theme-v2': theme });
  await ui.page.evaluate((t: string) => localStorage.setItem('ptah-theme-v2', t), theme);
}

/** Opens a chat tab bound to OLD_SESSION and returns its tab id. */
async function openBoundTab(ui: any, firstPrompt: string): Promise<string> {
  const page: Page = ui.page;
  await ui.goto('chat');
  const ta = page.locator('ptah-chat-input textarea[role="combobox"]').first();
  await ta.fill(firstPrompt);
  await page.locator('[data-testid="chat-send-btn"]').first().click();
  await expect.poll(async () => (await ui.getObservedCalls('chat:start')).length).toBe(1);
  const tabId = ((await ui.getObservedCalls('chat:start'))[0].params as { tabId: string }).tabId;
  await ui.pushEvent({ type: 'session:id-resolved', payload: { tabId, realSessionId: OLD_SESSION } });
  await ui.pushEvent(turnStateIdle(tabId, OLD_SESSION));
  await page.waitForTimeout(600);
  return tabId;
}

async function send(page: Page, text: string) {
  const ta = page.locator('ptah-chat-input textarea[role="combobox"]').first();
  await ta.fill(text);
  await page.locator('[data-testid="chat-send-btn"]').first().click();
}

async function delayCarryOver(ui: any) {
  await ui.app.evaluate(({ ipcMain }: any) => {
    const g = globalThis as any;
    const ls = ipcMain.listeners('rpc');
    ipcMain.removeAllListeners('rpc');
    ipcMain.on('rpc', (event: any, msg: any) => {
      const method = (msg?.payload || msg)?.method;
      if (method === 'chat:carry-over') {
        (g.__uiObservedCalls ?? []).push({ method, params: (msg?.payload || msg).params });
        setTimeout(() => ls.forEach((l: any) => l(event, msg)), 90000);
        return;
      }
      ls.forEach((l: any) => l(event, msg));
    });
  });
}

async function shoot(page: Page, el: Locator, name: string) {
  await page.screenshot({ path: `${OUT}/${name}-full.png` });
  await el.screenshot({ path: `${OUT}/${name}-closeup.png` });
}

async function baseMocks(ui: any, check: unknown, extra: Record<string, unknown> = {}) {
  await ui.mockRpc({
    'session:list': { sessions: [], total: 0, hasMore: false },
    'session:validate': { exists: true },
    'chat:start': { success: true },
    'chat:abort': { success: true },
    'chat:carry-over-check': check,
    ...extra,
  });
}

async function metrics(page: Page, banner: Locator) {
  return banner.evaluate((el: HTMLElement) => {
    const out: Record<string, unknown> = { banner: el.getBoundingClientRect().toJSON() };
    const btns = Array.from(el.querySelectorAll('button'));
    out['buttons'] = btns.map((b) => {
      const cs = getComputedStyle(b);
      const r = b.getBoundingClientRect();
      return { text: b.textContent?.trim(), w: r.width, h: r.height, disabled: b.disabled, ariaBusy: b.getAttribute('aria-busy'), color: cs.color, bg: cs.backgroundColor, border: cs.borderTopColor, fs: cs.fontSize, scrollW: b.scrollWidth, clientW: b.clientWidth };
    });
    const bs = getComputedStyle(el);
    out['bannerBg'] = bs.backgroundColor;
    out['hscroll'] = document.documentElement.scrollWidth > document.documentElement.clientWidth;
    return out;
  });
}
